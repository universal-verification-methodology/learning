// localparam_demo.v — parameter vs localparam
module fifo #(
  parameter WIDTH = 8
)(
  input  [WIDTH-1:0] d,
  output [WIDTH-1:0] q
);
  localparam DEPTH = WIDTH * 2;  // derived — not overridable
  assign q = d;
endmodule

module top;
  wire [3:0] din, dout;
  fifo #(.WIDTH(4)) u (.d(din), .q(dout));
endmodule
