// mux_styles.v — complete assign vs incomplete always
module mux_good(
  input  S, D0, D1,
  output Y
);
  assign Y = S ? D1 : D0;
endmodule

module mux_latch_risk(
  input  S, D0, D1,
  output reg Y
);
  always @(*) begin
    if (S) Y = D1;
    // no else — Y not assigned when S=0 → latch risk
  end
endmodule
